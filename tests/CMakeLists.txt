cmake_minimum_required(VERSION 3.16)
project(IndustrialRTOSHost LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/..")
add_library(industrial_logic "${ROOT}/components/control/control.cpp" "${ROOT}/components/diagnostics/diagnostics.cpp" "${ROOT}/components/telemetry/telemetry.cpp")
target_include_directories(industrial_logic PUBLIC "${ROOT}/components/control/include" "${ROOT}/components/diagnostics/include" "${ROOT}/components/telemetry/include")
target_compile_options(industrial_logic PRIVATE -Wall -Wextra -Wpedantic -Werror)
option(ENABLE_SANITIZERS "Address and undefined behavior sanitizers" OFF)
if(ENABLE_SANITIZERS)
  target_compile_options(industrial_logic PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(industrial_logic PUBLIC -fsanitize=address,undefined)
endif()
add_executable(control_tests control_tests.cpp)
target_link_libraries(control_tests PRIVATE industrial_logic)
target_compile_options(control_tests PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_test(NAME controller_contracts COMMAND control_tests)
add_executable(host_demo host_demo.cpp)
target_link_libraries(host_demo PRIVATE industrial_logic)
target_compile_options(host_demo PRIVATE -Wall -Wextra -Wpedantic -Werror)
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME json_demo_contract COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/verify_demo.py" $<TARGET_FILE:host_demo>)
